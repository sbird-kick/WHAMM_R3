3. WASI SHIM for setjmp/longjmp
-----------------------------------------------------------------
# Wizard implements no THROW opcode (wizard-engine CanonicalDefs.v3: "missing
# 0x08 THROW"), and LLVM's wasm SjLj lowering emits try/catch/throw that Wizard
# cannot parse. So we replace Lua's longjmp-based error unwinding: LUAI_TRY runs
# the protected body directly, LUAI_THROW aborts. Error-free / yield-free scripts
# never call luaD_throw, so abort() is never reached. (Consequence: coroutine
# .yield and error()-recovery are unsupported by these test binaries.)
SHIM="$JOBTMP/shim"; mkdir -p "$SHIM"
cat > "$SHIM/setjmp.h" <<'EOF'
#ifndef LUA_WASI_SETJMP_SHIM_H
#define LUA_WASI_SETJMP_SHIM_H
#include <stdlib.h>   /* abort */
typedef int jmp_buf[1];
#define LUAI_THROW(L,c)   abort()
#define LUAI_TRY(L,c,a)   { a }
#define luai_jmpbuf       int
#endif
EOF

# Lua core+libs. Defines:
#   _WASI_EMULATED_SIGNAL  -> sig_atomic_t (lstate.h) ; link -lwasi-emulated-signal
#   L_tmpnam=32            -> loslib.c os.tmpname buffer (never called)
#   luai_makeseed(L)=0     -> constant hash seed: determinism + avoids time()
"$CLANG" --target=wasm32-wasip1 -O2 -D_WASI_EMULATED_SIGNAL -DL_tmpnam=32 \
  '-Dluai_makeseed(L)=0' -I"$SHIM" -c "$SRC/lua_all.c" -o "$JOBTMP/lua_all.o"

# Deterministic stubs for os/io/time symbols wasi-libc omits or that must be
# constant (time() overridden so hash seed / math seed / table.sort pivot are
# deterministic and never hit wizeng's unimplemented clock_time_get).
"$CLANG" --target=wasm32-wasip1 -O2 -c "$REPO/lua_stubs.c" -o "$JOBTMP/lua_stubs.o"

5. BUILD one test .wasm  (repeat per lua_<name>.lua)
-----------------------------------------------------------------
# lua_runner.c reads an embedded byte array `lua_script` / `lua_script_len`.
# Generate that array from the .lua with xxd, then link. --gc-sections drops the
# io/os/debug library code that the reduced openlibs in lua_runner.c never
# references (shrinks the module and cuts whamm instrumentation cost).
build_one () {
  name="$1"                       # e.g. lua_sieve
  { echo "const unsigned char lua_script[] = {"; \
    xxd -i < "$REPO/$name.lua"; \
    echo "};"; \
    echo "const unsigned int lua_script_len = sizeof(lua_script);"; \
  } > "$JOBTMP/$name.gen.c"
  "$CLANG" --target=wasm32-wasip1 -O2 -Wl,--gc-sections \
    -I"$SRC" -I"$SHIM" \
    "$REPO/lua_runner.c" "$JOBTMP/$name.gen.c" \
    "$JOBTMP/lua_all.o" "$JOBTMP/lua_stubs.o" -lwasi-emulated-signal \
    -o "$REPO/$name.wasm"
}
